cmake_minimum_required(VERSION 3.22)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(amdva_mobileclip_example LANGUAGES C CXX)
  set(CMAKE_CXX_STANDARD 17)
  include(CTest)
  find_package(AMDVideoAnalytics REQUIRED)
endif()
find_package(PkgConfig REQUIRED)
pkg_check_modules(MC_JSON REQUIRED IMPORTED_TARGET json-glib-1.0)
amdva_use_host_glib(PkgConfig::MC_JSON)
set(AMDVA_ORT_ROOT "$ENV{VIRTUAL_ENV}/onnxruntime" CACHE PATH "Ryzen AI ONNX Runtime SDK root")
find_path(MC_ORT_INCLUDE onnxruntime_cxx_api.h HINTS "${AMDVA_ORT_ROOT}/include/onnxruntime" REQUIRED)
find_library(MC_ORT_LIBRARY NAMES onnxruntime HINTS "${AMDVA_ORT_ROOT}/lib" REQUIRED NO_DEFAULT_PATH)
find_program(MC_CARGO cargo REQUIRED)
set(MC_TOKENIZER_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../third_party/mobileclip-tokenizer")
set(MC_TOKENIZER "${CMAKE_CURRENT_BINARY_DIR}/tokenizer/release/libeg2_tokenizer.a")
add_custom_command(OUTPUT "${MC_TOKENIZER}"
  COMMAND ${MC_CARGO} build --locked --release --manifest-path "${MC_TOKENIZER_ROOT}/Cargo.toml" --target-dir "${CMAKE_CURRENT_BINARY_DIR}/tokenizer"
  DEPENDS "${MC_TOKENIZER_ROOT}/Cargo.toml" "${MC_TOKENIZER_ROOT}/Cargo.lock" "${MC_TOKENIZER_ROOT}/src/lib.rs" VERBATIM)
add_library(amdva_mobileclip SHARED runtime.cpp "${MC_TOKENIZER}")
add_library(AMDVA::amdva_mobileclip ALIAS amdva_mobileclip)
target_include_directories(amdva_mobileclip PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include> PRIVATE "${MC_ORT_INCLUDE}")
target_link_libraries(amdva_mobileclip PUBLIC AMDVA::amdva_inference AMDVA::amdva_runtime PRIVATE PkgConfig::MC_JSON "${MC_ORT_LIBRARY}" "${MC_TOKENIZER}" dl pthread m)
target_compile_options(amdva_mobileclip PRIVATE -Wall -Wextra)
target_link_options(amdva_mobileclip PRIVATE "LINKER:--exclude-libs,ALL")
set_target_properties(amdva_mobileclip PROPERTIES INSTALL_RPATH "${AMDVA_ORT_ROOT}/lib")
install(TARGETS amdva_mobileclip LIBRARY DESTINATION lib)
install(DIRECTORY include/ DESTINATION include)
install(FILES "${MC_TOKENIZER_ROOT}/LICENSE" DESTINATION share/licenses/amdva/mobileclip-tokenizer)
add_executable(amdva-mobileclip-interop interop.cpp)
target_link_libraries(amdva-mobileclip-interop PRIVATE amdva_mobileclip ${CMAKE_DL_LIBS})
install(TARGETS amdva-mobileclip-interop RUNTIME DESTINATION bin)
